-- 3. SCRIPT DE CAMERA LOCK
local fov3 = 50 
local prediction3 = 0.12 
local smoothness3 = 0.3 

local RunService3 = game:GetService("RunService")
local Players3 = game:GetService("Players")
local UserInputService3 = game:GetService("UserInputService")
local Cam3 = workspace.CurrentCamera
local LP3 = Players3.LocalPlayer

local function getTarget3()
    local nearest = nil
    local shortestDist = fov3
    local mousePos = UserInputService3:GetMouseLocation()

    for _, player in ipairs(Players3:GetPlayers()) do
        if player ~= LP3 and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local partName = _G.FullHS and "Head" or "HumanoidRootPart"
            local root = player.Character:FindFirstChild(partName)
            if root then
                local pos, onScreen = Cam3:WorldToViewportPoint(root.Position)
                if onScreen then
                    local dist = (Vector2.new(pos.X, pos.Y) - mousePos).Magnitude
                    if dist < shortestDist then
                        shortestDist = dist
                        nearest = player.Character
                    end
                end
            end
        end
    end
    return nearest
end

RunService3.RenderStepped:Connect(function()
    if _G.AimbotMode ~= "Camera Lock (Mob & PC)" then return end

    local targetChar = getTarget3()
    if targetChar then
        local partName = _G.FullHS and "Head" or "HumanoidRootPart"
        local targetPart = targetChar:FindFirstChild(partName)
        if targetPart then
            local predictedPos = targetPart.Position + (targetPart.Velocity * prediction3)
            local targetCFrame = CFrame.new(Cam3.CFrame.Position, predictedPos)
            Cam3.CFrame = Cam3.CFrame:Lerp(targetCFrame, smoothness3)
        end
    end
end)

-- 4. SCRIPT DE MOUSE LOCK (AIMBOT MOUSE)
local fov4 = 50 
local prediction4 = 0.12 
local smoothness4 = 0.3 
local lockedTarget4 = nil

local RunService4 = game:GetService("RunService")
local Players4 = game:GetService("Players")
local UserInputService4 = game:GetService("UserInputService")
local Cam4 = workspace.CurrentCamera

local function randomizeTargetPart4(character)
    if _G.FullHS then return character:FindFirstChild("Head") end
    if math.random(1, 100) <= 50 then
        return character:FindFirstChild("Head") or character:FindFirstChild("HumanoidRootPart")
    else
        return character:FindFirstChild("Torso") or character:FindFirstChild("HumanoidRootPart")
    end
end

local function getTargetUnderMouse4()
    local nearest = nil
    local shortestDist = math.huge
    local mousePos = UserInputService4:GetMouseLocation()

    for _, player in ipairs(Players4:GetPlayers()) do
        if player ~= Players4.LocalPlayer and player.Character then
            local part = player.Character:FindFirstChild("Torso") or player.Character:FindFirstChild("HumanoidRootPart")
            if part then
                local screenPos, onScreen = Cam4:WorldToViewportPoint(part.Position)
                if onScreen then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - Vector2.new(mousePos.X, mousePos.Y)).Magnitude
                    if dist < shortestDist and dist <= fov4 then
                        shortestDist = dist
                        nearest = player.Character
                    end
                end
            end
        end
    end
    return nearest
end

RunService4.RenderStepped:Connect(function()
    if _G.AimbotMode ~= "Mouse Lock (Mob & PC)" then return end
    
    local mousePos = UserInputService4:GetMouseLocation()
    local targetChar = getTargetUnderMouse4()
    
    if targetChar then
        if not lockedTarget4 or lockedTarget4.Parent ~= targetChar then
            lockedTarget4 = randomizeTargetPart4(targetChar)
        end
    else
        lockedTarget4 = nil
    end

    if lockedTarget4 and lockedTarget4.Parent and mousemoverel then
        local predictedPos = lockedTarget4.Position + (lockedTarget4.Velocity * prediction4)
        local targetPos = Cam4:WorldToViewportPoint(predictedPos)
        local delta = Vector2.new(targetPos.X, targetPos.Y) - Vector2.new(mousePos.X, mousePos.Y)
        mousemoverel(delta.X * smoothness4, delta.Y * smoothness4)
    end
end)

-- 5. SCRIPT DE ESP (VISUAL)
local RunService5 = game:GetService("RunService")
local Players5 = game:GetService("Players")
local Cam5 = workspace.CurrentCamera

local function createESP5(player)
    local highlight = Instance.new("Highlight")
    highlight.Enabled = false
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.FillTransparency = 0.5
    highlight.OutlineTransparency = 0.1

    local text = Drawing.new("Text")
    text.Visible = false
    text.Center = true
    text.Outline = true
    text.Font = Drawing.Fonts.UI
    text.Size = 13
    text.Color = Color3.fromRGB(255, 255, 255)

    local function applyToCharacter5(character)
        if character then
            highlight.Adornee = character
            pcall(function() highlight.Parent = game:GetService("CoreGui") end)
        end
    end

    player.CharacterAdded:Connect(applyToCharacter5)
    if player.Character then applyToCharacter5(player.Character) end

    local connection
    connection = RunService5.RenderStepped:Connect(function()
        if _G.ESPEnabled and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local head = player.Character:FindFirstChild("Head")
            if head then
                local headPos2d, onScreen = Cam5:WorldToViewportPoint(head.Position + Vector3.new(0, 1.5, 0))
                if onScreen then
                    text.Visible = true
                    text.Position = Vector2.new(headPos2d.X, headPos2d.Y)
                    text.Text = player.Name
                else text.Visible = false end
            else text.Visible = false end
            
            highlight.Enabled = true
            highlight.FillColor = player.TeamColor and player.TeamColor.Color or Color3.fromRGB(255,255,255)
            highlight.OutlineColor = highlight.FillColor
        else
            highlight.Enabled = false
            text.Visible = false
            if not player or not player.Parent then
                pcall(function() highlight:Destroy() text:Remove() end)
                connection:Disconnect()
            end
        end
    end)
end

for _, p in pairs(Players5:GetPlayers()) do
    if p ~= Players5.LocalPlayer then createESP5(p) end
end
Players5.PlayerAdded:Connect(function(p)
    if p ~= Players5.LocalPlayer then createESP5(p) end
end)
