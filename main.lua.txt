-- LÓGICA DO AIMBOT (BASEADA NO SEU CÓDIGO)
---------------------------------------------------------
local lockedTarget = nil

local function getTarget()
    local nearest = nil
    local shortestDist = _G.FOV
    local mousePos = UserInputService:GetMouseLocation()

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP and player.Character then
            if not _G.Wallbang then
                local parts = Cam:GetPartsObscuringTarget({player.Character.HumanoidRootPart.Position}, {LP.Character, player.Character})
                if #parts > 0 then continue end
            end

            local root = player.Character:FindFirstChild("HumanoidRootPart")
            if root then
                local pos, onScreen = Cam:WorldToViewportPoint(root.Position)
                if onScreen then
                    local dist = (Vector2.new(pos.X, pos.Y) - mousePos).Magnitude
                    if dist < shortestDist then
                        shortestDist = dist
                        nearest = player.Character
                    end
                end
            end
        end
    end
    return nearest
end

RunService.RenderStepped:Connect(function()
    -- Lógica Player (Speed/Jump/Noclip)
    if LP.Character and LP.Character:FindFirstChild("Humanoid") then
        LP.Character.Humanoid.WalkSpeed = _G.WalkSpeed
        LP.Character.Humanoid.JumpPower = _G.JumpPower
        if _G.NoClip then
            for _, part in pairs(LP.Character:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = false end
            end
        end
    end

    -- Lógica Aimbot
    if _G.AimbotMode == "Off" then return end
    
    local targetChar = getTarget()
    if targetChar then
        local partName = _G.FullHS and "Head" or (math.random(1,100) <= 50 and "Head" or "Torso")
        lockedTarget = targetChar:FindFirstChild(partName) or targetChar:FindFirstChild("HumanoidRootPart")
        
        if lockedTarget then
            local predictedPos = lockedTarget.Position + (lockedTarget.Velocity * _G.Prediction)
            
            if _G.AimbotMode == "Mouse Lock (Mob & PC)" then
                local sPos = Cam:WorldToViewportPoint(predictedPos)
                local mousePos = UserInputService:GetMouseLocation()
                if mousemoverel then
                    mousemoverel((sPos.X - mousePos.X) * _G.Smoothness, (sPos.Y - mousePos.Y) * _G.Smoothness)
                end
            elseif _G.AimbotMode == "Camera Lock (Mob & PC)" then
                Cam.CFrame = CFrame.new(Cam.CFrame.Position, predictedPos)
            end
        end
    end
end)

-- LÓGICA DO ESP (SEU CÓDIGO ORIGINAL MODIFICADO)
---------------------------------------------------------
local function createESP(player)
    local highlight = Instance.new("Highlight", game:GetService("CoreGui"))
    local text = Drawing.new("Text")
    text.Center, text.Outline, text.Size = true, true, 13

    RunService.RenderStepped:Connect(function()
        if _G.ESPEnabled and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = player.Character.HumanoidRootPart
            local pos, vis = Cam:WorldToViewportPoint(hrp.Position)

            if vis then
                highlight.Enabled = true
                highlight.Adornee = player.Character
                highlight.FillColor = (_G.ESPColorMode == "Team Color" and player.TeamColor) and player.TeamColor.Color or _G.CustomColor
                
                text.Visible = true
                text.Position = Vector2.new(pos.X, pos.Y - 30)
                local tool = player.Character:FindFirstChildOfClass("Tool")
                text.Text = player.Name .. (_G.ShowTools and (tool and "\n["..tool.Name.."]" or "\n[Nenhum]") or "")
                text.Color = highlight.FillColor
            else
                highlight.Enabled, text.Visible = false, false
            end
        else
            highlight.Enabled, text.Visible = false, false
        end
    end)
end
