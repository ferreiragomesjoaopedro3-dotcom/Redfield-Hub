-- // LOOP DE ATRIBUTOS DO HUMANÓIDE //
RunService.RenderStepped:Connect(function()
    pcall(function()
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
            local Hum = LocalPlayer.Character.Humanoid
            Hum.WalkSpeed = _G.WalkSpeedValue
            Hum.JumpPower = _G.JumpPowerValue
            Hum.UseJumpPower = true
        end
    end)
end)

-- // LOOP DE FISICA E COLISÃO (NOCLIP) //
RunService.Stepped:Connect(function()
    pcall(function()
        if _G.NoClipActive and LocalPlayer.Character then
            for _, Part in ipairs(LocalPlayer.Character:GetDescendants()) do
                if Part:IsA("BasePart") then
                    Part.CanCollide = false
                end
            end
        end
    end)
end)

-- // LÓGICA DE DETECÇÃO DE ALVO PARA AIMBOT //
local function GetClosestTarget()
    local Target = nil
    local MaxDist = _G.AimbotFOV
    local MousePos = UserInputService:GetMouseLocation()

    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer and Player.Character then
            local TargetPartName = _G.FullHS and "Head" or "HumanoidRootPart"
            local Part = Player.Character:FindFirstChild(TargetPartName)
            
            if Part then
                local ScreenPos, IsVisible = Camera:WorldToViewportPoint(Part.Position)
                if IsVisible then
                    local Distance = (Vector2.new(ScreenPos.X, ScreenPos.Y) - MousePos).Magnitude
                    if Distance < MaxDist then
                        MaxDist = Distance
                        Target = Part
                    end
                end
            end
        end
    end
    return Target
end

-- // LOOP DE CAMERA LOCK //
RunService.RenderStepped:Connect(function()
    if _G.AimbotMode == "Camera Lock (Mob & PC)" then
        local CurrentTarget = GetClosestTarget()
        if CurrentTarget then
            local PredictedPosition = CurrentTarget.Position + (CurrentTarget.Velocity * _G.AimbotPrediction)
            local LookAt = CFrame.new(Camera.CFrame.Position, PredictedPosition)
            Camera.CFrame = Camera.CFrame:Lerp(LookAt, _G.AimbotSmoothing)
        end
    end
end)

-- // LOOP DE MOUSE LOCK (MOUSEMOVEREL) //
RunService.RenderStepped:Connect(function()
    if _G.AimbotMode == "Mouse Lock (Mob & PC)" and mousemoverel then
        local CurrentTarget = GetClosestTarget()
        if CurrentTarget then
            local PredictedPosition = CurrentTarget.Position + (CurrentTarget.Velocity * _G.AimbotPrediction)
            local ScreenPos = Camera:WorldToViewportPoint(PredictedPosition)
            local MousePos = UserInputService:GetMouseLocation()
            
            local DeltaX = (ScreenPos.X - MousePos.X) * _G.AimbotSmoothing
            local DeltaY = (ScreenPos.Y - MousePos.Y) * _G.AimbotSmoothing
            mousemoverel(DeltaX, DeltaY)
        end
    end
end)

-- // LÓGICA DE ESP (RENDERIZAÇÃO DETALHADA) //
local function ImplementESP(TargetPlayer)
    local Highlight = Instance.new("Highlight")
    Highlight.Name = "Redfield_Highlight"
    Highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    Highlight.FillTransparency = 0.5
    Highlight.OutlineTransparency = 0
    Highlight.Parent = CoreGui

    local NameTag = Drawing.new("Text")
    NameTag.Visible = false
    NameTag.Center = true
    NameTag.Outline = true
    NameTag.Font = Drawing.Fonts.UI
    NameTag.Size = 14

    local ESPLoop
    ESPLoop = RunService.RenderStepped:Connect(function()
        if _G.ESPEnabled and TargetPlayer.Character and TargetPlayer.Character:FindFirstChild("HumanoidRootPart") then
            Highlight.Adornee = TargetPlayer.Character
            Highlight.Enabled = true
            Highlight.FillColor = _G.ESPColor
            Highlight.OutlineColor = Color3.fromRGB(255, 255, 255)

            local Head = TargetPlayer.Character:FindFirstChild("Head")
            if Head then
                local Vector, OnScreen = Camera:WorldToViewportPoint(Head.Position + Vector3.new(0, 2, 0))
                if OnScreen then
                    NameTag.Visible = true
                    NameTag.Position = Vector2.new(Vector.X, Vector.Y)
                    NameTag.Text = "[ " .. TargetPlayer.Name .. " ]"
                    NameTag.Color = _G.ESPColor
                else
                    NameTag.Visible = false
                end
            end
        else
            Highlight.Enabled = false
            NameTag.Visible = false
        end

        -- Auto-limpeza se o jogador sair
        if not TargetPlayer.Parent then
            Highlight:Destroy()
            NameTag:Remove()
            ESPLoop:Disconnect()
        end
    end)
end

-- // INICIALIZAÇÃO DO ESP PARA TODOS //
for _, p in pairs(Players:GetPlayers()) do
    if p ~= LocalPlayer then ImplementESP(p) end
end
Players.PlayerAdded:Connect(function(p)
    if p ~= LocalPlayer then ImplementESP(p) end
end)
