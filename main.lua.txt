-- LÓGICA DE FUNCIONAMENTO (ROBUSTA)

local RunService = game:GetService("RunService")
local Players = game:GetService("Players")
local LP = Players.LocalPlayer
local Cam = workspace.CurrentCamera
local UIS = game:GetService("UserInputService")

-- 1. SPEED & JUMP (Loop Infinito)
RunService.RenderStepped:Connect(function()
    if LP.Character and LP.Character:FindFirstChild("Humanoid") then
        LP.Character.Humanoid.WalkSpeed = _G.WS_Value
        LP.Character.Humanoid.JumpPower = _G.JP_Value
        LP.Character.Humanoid.UseJumpPower = true
    end
end)

-- 2. NOCLIP (Loop de Física)
RunService.Stepped:Connect(function()
    if _G.NoClipActive and LP.Character then
        for _, part in ipairs(LP.Character:GetDescendants()) do
            if part:IsA("BasePart") then part.CanCollide = false end
        end
    end
end)

-- 3. FUNÇÃO DE BUSCAR ALVO (UNIFICADA)
local function GetTarget()
    local target = nil
    local dist = _G.FOV_Radius
    local mouse = UIS:GetMouseLocation()
    
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local partName = _G.FullHS and "Head" or "HumanoidRootPart"
            local part = p.Character:FindFirstChild(partName)
            if part then
                local pos, vis = Cam:WorldToViewportPoint(part.Position)
                local mag = (Vector2.new(pos.X, pos.Y) - mouse).Magnitude
                if vis and mag < dist then
                    dist = mag
                    target = part
                end
            end
        end
    end
    return target
end

-- 4. AIMBOT (CAMERA E MOUSE)
RunService.RenderStepped:Connect(function()
    if _G.AimbotMode == "Off" then return end
    
    local target = GetTarget()
    if target then
        local prediction = target.Position + (target.Velocity * 0.12)
        
        if _G.AimbotMode == "Camera Lock (Mob & PC)" then
            Cam.CFrame = Cam.CFrame:Lerp(CFrame.new(Cam.CFrame.Position, prediction), 0.3)
        
        elseif _G.AimbotMode == "Mouse Lock (Mob & PC)" and mousemoverel then
            local sPos = Cam:WorldToViewportPoint(prediction)
            local mPos = UIS:GetMouseLocation()
            mousemoverel((sPos.X - mPos.X) * 0.3, (sPos.Y - mPos.Y) * 0.3)
        end
    end
end)

-- 5. ESP (CHAMS E NOME)
local function ApplyESP(p)
    local h = Instance.new("Highlight")
    h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    h.Parent = game:GetService("CoreGui")
    
    local t = Drawing.new("Text")
    t.Center = true; t.Outline = true; t.Size = 13
    
    RunService.RenderStepped:Connect(function()
        if _G.ESPEnabled and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            h.Adornee = p.Character
            h.Enabled = true
            h.FillColor = _G.ESPColor
            h.OutlineColor = Color3.new(1,1,1)
            
            local head = p.Character:FindFirstChild("Head")
            if head then
                local pos, vis = Cam:WorldToViewportPoint(head.Position + Vector3.new(0, 2, 0))
                t.Visible = vis
                t.Position = Vector2.new(pos.X, pos.Y)
                t.Text = p.Name
                t.Color = _G.ESPColor
            end
        else
            h.Enabled = false; t.Visible = false
        end
        if not p.Parent then h:Destroy(); t:Remove() end
    end)
end

for _, p in pairs(Players:GetPlayers()) do if p ~= LP then ApplyESP(p) end end
Players.PlayerAdded:Connect(function(p) if p ~= LP then ApplyESP(p) end end)
