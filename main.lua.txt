-- // CORE LOGIC: AIMBOT (SEM COMPROMISSO) //
local function getBestTarget()
    local target = nil
    local dist = _G.FOV
    local mouse = UserInputService:GetMouseLocation()

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local char = p.Character
            local root = char.HumanoidRootPart
            
            -- Lógica de Wallbang (Check de Visibilidade)
            if not _G.Wallbang then
                local ray = Ray.new(Cam.CFrame.Position, (root.Position - Cam.CFrame.Position).Unit * 1000)
                local hit = workspace:FindPartOnRayWithIgnoreList(ray, {LP.Character, char})
                if hit then continue end
            end

            local pos, vis = Cam:WorldToViewportPoint(root.Position)
            if vis then
                local magnitude = (Vector2.new(pos.X, pos.Y) - mouse).Magnitude
                if magnitude < dist then
                    dist = magnitude
                    target = char
                end
            end
        end
    end
    return target
end

-- // LOOP DE ATUALIZAÇÃO CONSTANTE //
RunService.RenderStepped:Connect(function()
    -- Estatísticas do Player
    if LP.Character and LP.Character:FindFirstChild("Humanoid") then
        LP.Character.Humanoid.WalkSpeed = _G.WalkSpeed
        LP.Character.Humanoid.JumpPower = _G.JumpPower
        LP.Character.Humanoid.UseJumpPower = true
        
        -- NoClip Ativo (Pesado)
        if _G.NoClip then
            for _, v in pairs(LP.Character:GetDescendants()) do
                if v:IsA("BasePart") then v.CanCollide = false end
            end
        end
    end

    -- Aimbot Execution
    if _G.AimbotMode ~= "Off" then
        local targetChar = getBestTarget()
        if targetChar then
            local part = _G.FullHS and targetChar:FindFirstChild("Head") or (math.random(1,2) == 1 and targetChar:FindFirstChild("Head") or targetChar:FindFirstChild("HumanoidRootPart"))
            
            if part then
                local predicted = part.Position + (part.Velocity * _G.Prediction)
                
                if _G.AimbotMode == "Mouse Lock (Mob & PC)" then
                    local screenPos = Cam:WorldToViewportPoint(predicted)
                    local mPos = UserInputService:GetMouseLocation()
                    if mousemoverel then
                        mousemoverel((screenPos.X - mPos.X) * _G.Smoothness, (screenPos.Y - mPos.Y) * _G.Smoothness)
                    end
                elseif _G.AimbotMode == "Camera Lock (Mob & PC)" then
                    Cam.CFrame = CFrame.new(Cam.CFrame.Position, predicted)
                end
            end
        end
    end
end)

-- // ESP SYSTEM (DRAWING + HIGHLIGHT) //
local function applyESP(p)
    local highlight = Instance.new("Highlight", game:GetService("CoreGui"))
    local label = Drawing.new("Text")
    label.Center = true
    label.Outline = true
    label.Size = 14

    RunService.RenderStepped:Connect(function()
        if _G.ESPEnabled and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = p.Character.HumanoidRootPart
            local pos, vis = Cam:WorldToViewportPoint(hrp.Position)

            if vis then
                highlight.Enabled = true
                highlight.Adornee = p.Character
                local color = (_G.ESPColorMode == "Team Color" and p.TeamColor) and p.TeamColor.Color or _G.CustomColor
                highlight.FillColor = color
                highlight.OutlineColor = Color3.new(1,1,1)
                
                label.Visible = true
                label.Position = Vector2.new(pos.X, pos.Y - 40)
                label.Color = color
                
                local tool = p.Character:FindFirstChildOfClass("Tool")
                local info = p.Name
                if _G.ShowTools then
                    info = info .. "\n[" .. (tool and tool.Name or "Mãos Vazias") .. "]"
                end
                label.Text = info
            else
                highlight.Enabled = false
                label.Visible = false
            end
        else
            highlight.Enabled = false
            label.Visible = false
        end
    end)
end

for _, p in pairs(Players:GetPlayers()) do if p ~= LP then applyESP(p) end end
